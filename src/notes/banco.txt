create database ipsa_estoque;

use ipsa_estoque;

create table usuario(
usu_id int not null auto_increment,
usu_nome varchar(30),
usu_email varchar(30),
usu_senha varchar(30),
usu_nivel varchar(30),
primary key (usu_id));

create table equipamento(
equ_id int not null auto_increment,
equ_nome varchar(30),
equ_tipo varchar(30),
equ_modelo varchar(30),
equ_descricao varchar(30),
equ_data_aquisicao varchar(30),
equ_quantidade varchar(30),
equ_status varchar(30),
equ_foto varchar(30),
equ_categoria varchar(30),
primary key (equ_id));

create table emprestimo(
emp_id date not null,
emp_descricao int not null,
usu_nome int not null,
emp_data_retirada double not null,
emp_previsao_devolucao double not null,
emp_data_devolucao double not null,
emp_evento double not null,
emp_status double not null,
primary key (emp_data),
foreign key (usu_nome) references usuario (usu_nome) on delete cascade on update cascade;
