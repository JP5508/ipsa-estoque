user: id, nome, email, senha, nivel (2-3 niveis?)
categoria: id, nome, descrição
equipamento: id, nome, tipo, modelo, descrição, data de aquisição, valor de aquisição, quantidade, status (disponível, manutenção, emprestado, usando, programado), foto?, id_categoria
emprestimo: id, descrição, id_usuario, data da retirada, data prevista da devolução, data de devolução, evento(campo opcional mais pra saber se é pra algum evento ou se foi emprestado pra alguém), status(programado, emprestado ,atrasado, devolvido)

create database ipsa_estoque;

use ipsa_estoque;

CREATE TABLE usuario (
usu_id INT NOT NULL AUTO_INCREMENT,
usu_nome VARCHAR(100) NOT NULL,
usu_email VARCHAR(255) NOT NULL,
usu_senha VARCHAR(255) NOT NULL,
usu_nivel VARCHAR(1) NOT NULL,
PRIMARY KEY (usu_id),
UNIQUE (usu_email));

CREATE TABLE equipamento (
equ_id INT NOT NULL AUTO_INCREMENT,
equ_nome VARCHAR(50) NOT NULL,
equ_tipo VARCHAR(100),
equ_modelo VARCHAR(50),
equ_descricao TEXT,
equ_data_aquisicao DATE,
equ_quantidade INT UNSIGNED NOT NULL,
equ_status VARCHAR(10) NOT NULL,
equ_categoria VARCHAR(50),
equ_foto BLOB,
PRIMARY KEY (equ_id));

CREATE TABLE emprestimo (
emp_id INT NOT NULL AUTO_INCREMENT,
emp_descricao TEXT,
usu_id INT NOT NULL,
equ_id INT NOT NULL,
emp_data_retirada DATETIME NOT NULL,
emp_previsao_devolucao DATETIME NOT NULL,
emp_data_devolucao DATETIME DEFAULT NULL,
emp_evento VARCHAR(100),
emp_status VARCHAR(10) NOT NULL,

PRIMARY KEY (emp_id),

FOREIGN KEY (usu_id) REFERENCES usuario (usu_id)

FOREIGN KEY (equ_id) REFERENCES equipamento (equ_id)
